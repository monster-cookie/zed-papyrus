Int Function Evaluate(Int value)
    Int result = value
    result = 1
    result += 2
    result -= 3
    result *= 4
    result /= 2
    result %= 2

    If result > 0
        Debug.Trace("positive")
    ElseIf result < 0
        Debug.Trace("negative")
    Else
        Debug.Trace("zero")
    EndIf

    While result < 10
        result += 1
    EndWhile

    Return result
EndFunction
